import pandas as pd
import numpy as np

def make_predictions(
    trained_classifier, 
    feature_encoder, 
    scaler, 
    feature_engineered_data: pd.DataFrame,
    parameters: dict
) -> pd.DataFrame:
    """Make predictions on the entire dataset"""
    
    # Prepare features for prediction
    numerical_features = parameters["numerical_features"]
    categorical_features = parameters["categorical_features"]
    target_column = parameters["target_column"]
    probability_threshold = parameters["probability_threshold"]
    
    all_features = numerical_features + categorical_features
    available_features = [f for f in all_features if f in feature_engineered_data.columns]
    
    X_pred = feature_engineered_data[available_features].copy()
    
    # Encode categorical features
    for feature in categorical_features:
        if feature in X_pred.columns and feature in feature_encoder:
            encoder = feature_encoder[feature]
            X_pred[feature] = encoder.transform(X_pred[feature].astype(str))
    
    # Scale numerical features
    for feature in numerical_features:
        if feature in X_pred.columns:
            X_pred[feature] = scaler.transform(X_pred[[feature]])
    
    # Make predictions
    predictions = trained_classifier.predict(X_pred)
    probabilities = trained_classifier.predict_proba(X_pred)
    
    # Create results dataframe
    results = feature_engineered_data.copy()
    results['predicted_churn_risk'] = predictions
    results['prediction_probability'] = np.max(probabilities, axis=1)
    
    # Add probability for each class
    for i, class_label in enumerate(trained_classifier.classes_):
        results[f'probability_class_{class_label}'] = probabilities[:, i]
    
    # Add business interpretation
    risk_mapping = {0: 'Low Risk', 1: 'Medium Risk', 2: 'High Risk'}
    results['risk_category'] = results['predicted_churn_risk'].map(risk_mapping)
    
    return results

def save_predictions(predictions: pd.DataFrame) -> pd.DataFrame:
    """Save predictions with relevant columns only"""
    
    # Select key columns for final output
    output_columns = [
        'customer_id', 'age', 'annual_income', 'credit_score', 'account_balance',
        'transaction_frequency', 'fraudulent_transactions_count',
        'predicted_churn_risk', 'prediction_probability', 'risk_category'
    ]
    
    # Only include columns that exist
    available_columns = [col for col in output_columns if col in predictions.columns]
    final_predictions = predictions[available_columns]
    
    print(f"Predictions completed for {len(final_predictions)} customers")
    print("\nRisk Category Distribution:")
    print(final_predictions['risk_category'].value_counts())
    
    return final_predictions
