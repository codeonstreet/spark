import pandas as pd
import numpy as np
from pyspark.sql import SparkSession
from pyspark.sql.types import *
from pyspark.sql import functions as F
from datetime import datetime, timedelta
import os

def generate_customer_data_spark(num_customers: int) -> str:
    """Generate customer data using PySpark spark.range and save as distributed Parquet"""
    
    print("🚀 Generating customer data with PySpark...")
    
    # Initialize Spark session
    spark = SparkSession.builder \
        .appName("CustomerDataGeneration") \
        .config("spark.sql.adaptive.enabled", "true") \
        .getOrCreate()
    
    # Define choices for categorical variables
    genders = ["Male", "Female"]
    employment_statuses = ["Employed", "Self-Employed", "Unemployed", "Retired"]
    education_levels = ["High School", "Bachelor", "Master", "PhD"]
    marital_statuses = ["Single", "Married", "Divorced", "Widowed"]
    
    # Generate base DataFrame with spark.range
    customers_df = spark.range(num_customers) \
        .withColumn("customer_id", F.concat(F.lit("CUST_"), F.lpad(F.col("id"), 8, "0"))) \
        .withColumn("age", (F.rand() * 62 + 18).cast("integer")) \
        .withColumn("gender", F.element_at(F.array([F.lit(g) for g in genders]), (F.rand() * len(genders) + 1).cast("integer"))) \
        .withColumn("annual_income", F.round(F.rand() * 180000 + 20000, 2)) \
        .withColumn("credit_score", (F.rand() * 550 + 300).cast("integer")) \
        .withColumn("account_balance", F.round(F.rand() * 99000 + 1000, 2)) \
        .withColumn("employment_status", F.element_at(F.array([F.lit(e) for e in employment_statuses]), (F.rand() * len(employment_statuses) + 1).cast("integer"))) \
        .withColumn("education_level", F.element_at(F.array([F.lit(e) for e in education_levels]), (F.rand() * len(education_levels) + 1).cast("integer"))) \
        .withColumn("marital_status", F.element_at(F.array([F.lit(m) for m in marital_statuses]), (F.rand() * len(marital_statuses) + 1).cast("integer"))) \
        .withColumn("num_dependents", (F.rand() * 6).cast("integer")) \
        .withColumn("years_with_bank", (F.rand() * 21).cast("integer")) \
        .withColumn("total_products", (F.rand() * 5 + 1).cast("integer")) \
        .drop("id")
    
    # Save as distributed Parquet files
    output_path = "data/01_raw/customers"
    
    # Clean up existing directory
    if os.path.exists(output_path):
        import shutil
        shutil.rmtree(output_path)
    
    # Write as distributed Parquet - let Spark decide optimal number of files
    customers_df.write.mode("overwrite").parquet(output_path)
    
    print(f"✅ Generated {num_customers} customer records")
    print(f"💾 Saved to: {output_path}")
    print(f"📊 Customer data schema: {customers_df.schema}")
    
    # Show statistics
    print("📈 Customer Data Summary:")
    customers_df.select([
        F.count("*").alias("total_customers"),
        F.avg("annual_income").alias("avg_income"),
        F.avg("credit_score").alias("avg_credit_score"),
        F.avg("account_balance").alias("avg_balance")
    ]).show()
    
    # Show distribution
    print("👥 Gender Distribution:")
    customers_df.groupBy("gender").count().show()
    
    spark.stop()
    
    return output_path

def generate_transaction_data_spark(num_transactions: int, start_date: str, end_date: str) -> str:
    """Generate transaction data using PySpark spark.range and save as distributed Parquet"""
    
    print("🚀 Generating transaction data with PySpark...")
    
    spark = SparkSession.builder \
        .appName("TransactionDataGeneration") \
        .config("spark.sql.adaptive.enabled", "true") \
        .getOrCreate()
    
    # Read customer data to get customer IDs
    customers_path = "data/01_raw/customers"
    customers_df = spark.read.parquet(customers_path)
    customer_count = customers_df.count()
    
    print(f"📖 Using {customer_count} customers for transaction generation")
    
    # Define choices for categorical variables
    transaction_types = ["Purchase", "Withdrawal", "Transfer", "Deposit"]
    merchant_categories = ["Retail", "Groceries", "Entertainment", "Utilities", "Travel"]
    locations = ["New York", "California", "Texas", "Florida", "Illinois"]
    
    # Convert date strings to timestamps
    start_ts = int(datetime.strptime(start_date, "%Y-%m-%d").timestamp())
    end_ts = int(datetime.strptime(end_date, "%Y-%m-%d").timestamp())
    
    # Generate transactions using spark.range
    transactions_df = spark.range(num_transactions) \
        .withColumn("transaction_id", F.concat(F.lit("TXN_"), F.lpad(F.col("id"), 10, "0"))) \
        .withColumn("customer_id", 
                   F.concat(F.lit("CUST_"), 
                           F.lpad((F.rand() * customer_count).cast("integer"), 8, "0"))) \
        .withColumn("transaction_date", 
                   F.from_unixtime(F.rand() * (end_ts - start_ts) + start_ts)) \
        .withColumn("transaction_amount", F.round(F.rand() * 4999 + 1, 2)) \
        .withColumn("transaction_type", 
                   F.element_at(F.array([F.lit(t) for t in transaction_types]), 
                               (F.rand() * len(transaction_types) + 1).cast("integer"))) \
        .withColumn("merchant_category", 
                   F.element_at(F.array([F.lit(m) for m in merchant_categories]), 
                               (F.rand() * len(merchant_categories) + 1).cast("integer"))) \
        .withColumn("location", 
                   F.element_at(F.array([F.lit(l) for l in locations]), 
                               (F.rand() * len(locations) + 1).cast("integer"))) \
        .withColumn("is_fraud", 
                   F.when(F.rand() < 0.02, 1).otherwise(0).cast("integer")) \
        .drop("id")
    
    # Save as distributed Parquet files
    output_path = "data/01_raw/transactions"
    
    # Clean up existing directory
    if os.path.exists(output_path):
        import shutil
        shutil.rmtree(output_path)
    
    # Write as distributed Parquet - let Spark decide optimal number of files
    transactions_df.write.mode("overwrite").parquet(output_path)
    
    print(f"✅ Generated {num_transactions} transaction records")
    print(f"💾 Saved to: {output_path}")
    print(f"📊 Transaction data schema: {transactions_df.schema}")
    
    # Show statistics
    print("📈 Transaction Data Summary:")
    transactions_df.select([
        F.count("*").alias("total_transactions"),
        F.avg("transaction_amount").alias("avg_amount"),
        F.sum("is_fraud").alias("fraudulent_transactions"),
        F.countDistinct("customer_id").alias("unique_customers")
    ]).show()
    
    # Show distributions
    print("💰 Transaction Type Distribution:")
    transactions_df.groupBy("transaction_type") \
        .count() \
        .orderBy("count", ascending=False) \
        .show()
    
    print("🏪 Merchant Category Distribution:")
    transactions_df.groupBy("merchant_category") \
        .count() \
        .orderBy("count", ascending=False) \
        .show()
    
    spark.stop()
    
    return output_path

def process_customer_data() -> pd.DataFrame:
    """Read Parquet directory with pandas and process customer data"""
    
    input_path = "data/01_raw/customers"
    print(f"📖 Reading customer data from {input_path}")
    
    # Read Parquet directory using pandas
    customers_df = pd.read_parquet(input_path)
    
    print(f"🔧 Processing {len(customers_df)} customer records")
    
    # Remove duplicates
    initial_count = len(customers_df)
    customers_processed = customers_df.drop_duplicates(subset=['customer_id'])
    final_count = len(customers_processed)
    
    if initial_count != final_count:
        print(f"⚠️  Removed {initial_count - final_count} duplicate customer records")
    
    # Handle missing values
    customers_processed['credit_score'] = customers_processed['credit_score'].fillna(
        customers_processed['credit_score'].median()
    )
    customers_processed['annual_income'] = customers_processed['annual_income'].fillna(
        customers_processed['annual_income'].median()
    )
    
    print(f"✅ Processed {len(customers_processed)} customer records")
    
    return customers_processed

def process_transaction_data() -> pd.DataFrame:
    """Read Parquet directory with pandas and process transaction data"""
    
    input_path = "data/01_raw/transactions"
    print(f"📖 Reading transaction data from {input_path}")
    
    # Read Parquet directory using pandas
    transactions_df = pd.read_parquet(input_path)
    
    print(f"🔧 Processing {len(transactions_df)} transaction records")
    
    # Remove duplicates
    initial_count = len(transactions_df)
    transactions_processed = transactions_df.drop_duplicates(subset=['transaction_id'])
    final_count = len(transactions_processed)
    
    if initial_count != final_count:
        print(f"⚠️  Removed {initial_count - final_count} duplicate transaction records")
    
    # Handle missing values
    transactions_processed['transaction_amount'] = transactions_processed['transaction_amount'].fillna(0)
    transactions_processed['is_fraud'] = transactions_processed['is_fraud'].fillna(0)
    
    # Convert transaction_date to datetime if it's not already
    if not pd.api.types.is_datetime64_any_dtype(transactions_processed['transaction_date']):
        transactions_processed['transaction_date'] = pd.to_datetime(transactions_processed['transaction_date'])
    
    print(f"✅ Processed {len(transactions_processed)} transaction records")
    
    return transactions_processed
