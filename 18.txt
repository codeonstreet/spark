import pandas as pd
import numpy as np
from pyspark.sql import SparkSession
from pyspark.sql.types import *
from pyspark.sql import functions as F
from datetime import datetime, timedelta
import random
import os

def generate_customer_data_spark(num_customers: int) -> pd.DataFrame:
    """Generate customer data using PySpark and return as pandas DataFrame"""
    
    print("🚀 Generating customer data with PySpark...")
    
    # Initialize Spark session
    spark = SparkSession.builder \
        .appName("CustomerDataGeneration") \
        .config("spark.sql.adaptive.enabled", "true") \
        .getOrCreate()
    
    # Generate data
    customer_data = []
    for i in range(num_customers):
        customer_id = f"CUST_{i:08d}"
        age = random.randint(18, 80)
        gender = random.choice(["Male", "Female"])
        annual_income = round(random.uniform(20000, 200000), 2)
        credit_score = random.randint(300, 850)
        account_balance = round(random.uniform(1000, 100000), 2)
        employment_status = random.choice(["Employed", "Self-Employed", "Unemployed", "Retired"])
        education_level = random.choice(["High School", "Bachelor", "Master", "PhD"])
        marital_status = random.choice(["Single", "Married", "Divorced", "Widowed"])
        num_dependents = random.randint(0, 5)
        years_with_bank = random.randint(0, 20)
        total_products = random.randint(1, 6)
        
        customer_data.append((
            customer_id, age, gender, annual_income, credit_score,
            account_balance, employment_status, education_level,
            marital_status, num_dependents, years_with_bank, total_products
        ))
    
    # Create Spark DataFrame
    customer_schema = StructType([
        StructField("customer_id", StringType(), True),
        StructField("age", IntegerType(), True),
        StructField("gender", StringType(), True),
        StructField("annual_income", DoubleType(), True),
        StructField("credit_score", IntegerType(), True),
        StructField("account_balance", DoubleType(), True),
        StructField("employment_status", StringType(), True),
        StructField("education_level", StringType(), True),
        StructField("marital_status", StringType(), True),
        StructField("num_dependents", IntegerType(), True),
        StructField("years_with_bank", IntegerType(), True),
        StructField("total_products", IntegerType(), True)
    ])
    
    customer_df = spark.createDataFrame(customer_data, customer_schema)
    
    # Convert to pandas DataFrame (let Kedro handle the file saving)
    pandas_df = customer_df.toPandas()
    
    print(f"✅ Generated {len(pandas_df)} customer records")
    print(f"📊 Sample data:")
    print(pandas_df.head(3))
    
    spark.stop()
    
    return pandas_df

def generate_transaction_data_spark(num_transactions: int, start_date: str, end_date: str, raw_customers: pd.DataFrame) -> pd.DataFrame:
    """Generate transaction data using PySpark and return as pandas DataFrame"""
    
    print("🚀 Generating transaction data with PySpark...")
    
    spark = SparkSession.builder \
        .appName("TransactionDataGeneration") \
        .config("spark.sql.adaptive.enabled", "true") \
        .getOrCreate()
    
    # Get customer IDs from the pandas DataFrame
    customer_ids = raw_customers['customer_id'].tolist()
    
    print(f"📖 Using {len(customer_ids)} customer IDs for transaction generation")
    
    # Generate transaction data
    transaction_data = []
    start_dt = datetime.strptime(start_date, "%Y-%m-%d")
    end_dt = datetime.strptime(end_date, "%Y-%m-%d")
    
    for i in range(num_transactions):
        transaction_id = f"TXN_{i:010d}"
        customer_id = random.choice(customer_ids)
        
        # Generate random date between start and end
        random_days = random.randint(0, (end_dt - start_dt).days)
        random_seconds = random.randint(0, 86400)
        transaction_date = start_dt + timedelta(days=random_days, seconds=random_seconds)
        
        transaction_amount = round(random.uniform(1, 5000), 2)
        transaction_type = random.choice(["Purchase", "Withdrawal", "Transfer", "Deposit"])
        merchant_category = random.choice(["Retail", "Groceries", "Entertainment", "Utilities", "Travel"])
        location = random.choice(["New York", "California", "Texas", "Florida", "Illinois"])
        is_fraud = random.choices([0, 1], weights=[0.98, 0.02])[0]
        
        transaction_data.append((
            transaction_id, customer_id, transaction_date, transaction_amount,
            transaction_type, merchant_category, location, is_fraud
        ))
    
    # Create Spark DataFrame
    transaction_schema = StructType([
        StructField("transaction_id", StringType(), True),
        StructField("customer_id", StringType(), True),
        StructField("transaction_date", TimestampType(), True),
        StructField("transaction_amount", DoubleType(), True),
        StructField("transaction_type", StringType(), True),
        StructField("merchant_category", StringType(), True),
        StructField("location", StringType(), True),
        StructField("is_fraud", IntegerType(), True)
    ])
    
    transaction_df = spark.createDataFrame(transaction_data, transaction_schema)
    
    # Convert to pandas DataFrame (let Kedro handle the file saving)
    pandas_df = transaction_df.toPandas()
    
    print(f"✅ Generated {len(pandas_df)} transaction records")
    print(f"📊 Sample data:")
    print(pandas_df.head(3))
    
    # Show some statistics
    fraud_count = pandas_df['is_fraud'].sum()
    avg_amount = pandas_df['transaction_amount'].mean()
    print(f"📈 Stats - Fraudulent transactions: {fraud_count}, Avg amount: ${avg_amount:.2f}")
    
    spark.stop()
    
    return pandas_df

def process_customer_data(raw_customers: pd.DataFrame) -> pd.DataFrame:
    """Process customer data using pandas"""
    
    print(f"🔧 Processing {len(raw_customers)} customer records")
    
    # Create a copy to avoid modifying the original
    customers_processed = raw_customers.copy()
    
    # Remove duplicates
    initial_count = len(customers_processed)
    customers_processed = customers_processed.drop_duplicates(subset=['customer_id'])
    final_count = len(customers_processed)
    
    if initial_count != final_count:
        print(f"⚠️  Removed {initial_count - final_count} duplicate customer records")
    
    # Handle missing values
    customers_processed['credit_score'] = customers_processed['credit_score'].fillna(
        customers_processed['credit_score'].median()
    )
    customers_processed['annual_income'] = customers_processed['annual_income'].fillna(
        customers_processed['annual_income'].median()
    )
    
    print(f"✅ Processed {len(customers_processed)} customer records")
    
    return customers_processed

def process_transaction_data(raw_transactions: pd.DataFrame) -> pd.DataFrame:
    """Process transaction data using pandas"""
    
    print(f"🔧 Processing {len(raw_transactions)} transaction records")
    
    # Create a copy to avoid modifying the original
    transactions_processed = raw_transactions.copy()
    
    # Remove duplicates
    initial_count = len(transactions_processed)
    transactions_processed = transactions_processed.drop_duplicates(subset=['transaction_id'])
    final_count = len(transactions_processed)
    
    if initial_count != final_count:
        print(f"⚠️  Removed {initial_count - final_count} duplicate transaction records")
    
    # Handle missing values
    transactions_processed['transaction_amount'] = transactions_processed['transaction_amount'].fillna(0)
    transactions_processed['is_fraud'] = transactions_processed['is_fraud'].fillna(0)
    
    # Convert transaction_date to datetime if it's not already
    if not pd.api.types.is_datetime64_any_dtype(transactions_processed['transaction_date']):
        transactions_processed['transaction_date'] = pd.to_datetime(transactions_processed['transaction_date'])
    
    print(f"✅ Processed {len(transactions_processed)} transaction records")
    
    return transactions_processed
